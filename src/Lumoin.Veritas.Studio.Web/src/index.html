PREFIX bat: <https://veritas.app/ns/battery#>
PREFIX dpp: <https://veritas.app/ns/dpp#>

# EV batteries whose recycled cobalt is below the 16% target —
# the recycled-content claim lives on the supplier's endpoint.
SELECT ?battery ?cathode ?recycledCobalt ?passport
WHERE {
  ?battery a bat:Battery ;
           bat:cathode  ?cathode ;
           dpp:passport ?passport .
  SERVICE <https://supplier.example/dpp/sparql> {   # across a trust boundary, access-gated
    ?cathode  dpp:recycledContent ?content .
    ?content  dpp:substance "cobalt" ; dpp:share ?recycledCobalt .
  }
  FILTER ( ?recycledCobalt < 16 )
}
ORDER BY ASC(?recycledCobalt)# RecycledCobaltShape — every EV battery declares a cobalt recycled-content share

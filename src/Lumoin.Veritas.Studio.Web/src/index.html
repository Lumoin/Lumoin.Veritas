# lives on the brand's federated facility registry, access-gated.
SELECT ?garment ?facility ?riskScore
WHERE {
  ?garment a untp:Product ; untp:productCategory "6205" .
  ?garment untp:upstreamInput+ ?upstream .
  ?upstream untp:producedAtFacility ?facility .
  SERVICE <https://registry.example/water/sparql> {   # across a trust boundary, access-gated
    ?facility vw:locatedInBasin ?basin .
    ?basin    vw:baselineWaterStress ?riskScore .
  }
  FILTER ( ?riskScore >= 4.0 )
}

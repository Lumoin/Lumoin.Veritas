PREFIX bat: <https://veritas.app/ns/battery#>
PREFIX mat: <https://veritas.app/ns/materials#>

# Battery packs below the recycled-cobalt threshold whose cathode
# draws on a scarce metal — the metal grades live on a federated
# materials registry, access-gated.
SELECT ?battery ?cathode ?scarcity
WHERE {
  ?battery a bat:Battery ; bat:cathode ?cathode ; bat:recycledCobalt ?recycled .
  SERVICE <https://registry.example/materials/sparql> {   # across a trust boundary, access-gated
    ?cathode mat:primaryMetal ?metal .
    ?metal   mat:scarcityIndex ?scarcity .
  }
  FILTER ( ?scarcity >= 4.0 && ?recycled < 10 )
}
ORDER BY DESC(?scarcity)# RecycledContentShape — every battery declares its recycled cobalt share
